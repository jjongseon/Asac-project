def fetch_kickstarter_projects(next_cursor=None):
    url = "https://www.kickstarter.com/graph"
   
    # 2. 제공해주신 GraphQL 쿼리
    query = """
    query($nextCursor: String) {
      projects(recommended: true, first: 6, after: $nextCursor, state: LIVE) {
        edges {
          node {
            id
            pid
            name
            url
            percentFunded
            backersCount
            pledged { amount }
            category { name }
            profile { blurb }
          }
          cursor
        }
      }
    }
    """
   
    payload = {
        "query": query,
        "variables": {"nextCursor": next_cursor}
    }
   
    response = requests.post(url, headers=headers, json=payload)
    return response.json()

# 3. 롤링 실행 로직
all_projects = []

print("데이터 수집 시작...")
for page in range(1, 6):  # 5페이지까지 반복
    data = fetch_kickstarter_projects(current_cursor)
   
    try:
        edges = data['data']['projects']['edges']
        if not edges:
            break
           
        all_projects.extend(edges)
        # 다음 페이지를 위해 마지막 커서 업데이트
        current_cursor = edges[-1]['cursor']
        print(f"페이지 {page} 완료: {len(edges)}개 프로젝트 수집 (다음 커서: {current_cursor})")
       
    except (KeyError, TypeError) as e:
        print("데이터 추출 중 에러 발생:", data)
        break
